\cvsection{Selected research projects}
\descriptionstyle{
\begin{itemize}[leftmargin=1.5em, itemsep=4pt, topsep=3pt, parsep=0pt]
  \item \textbf{BALEEN: RNA modification detection.} Developed a method using dynamic time warping and Bayesian modelling to identify RNA modification sites from nanopore direct RNA sequencing signals.
  \item \textbf{TAPE: tissue-adaptive transcriptome deconvolution.} Analysed and interpreted single-sample gene set enrichment analysis (ssGSEA) results and contributed to manuscript writing. The study combines bulk and single-cell RNA-seq through a deep autoencoder and was published in \textit{Nature Communications} (2022).
  \item \textbf{FIN: gene fusion detection.} Worked on a Transformer-based model for gene fusion detection.
  \item \textbf{Medaka responses to hypoxia.} Contributed to integrative miRNA--mRNA analysis of embryonic development and sex determination under hypoxia, published in \textit{Frontiers in Marine Science} (2022).
\end{itemize}
}
