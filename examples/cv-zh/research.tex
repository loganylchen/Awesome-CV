\cvsection{代表性研究项目}
\descriptionstyle{
\begin{itemize}[leftmargin=1.5em, itemsep=4pt, topsep=3pt, parsep=0pt]
  \item \textbf{BALEEN：RNA 修饰检测。}开发结合动态时间规整与贝叶斯模型的分析方法，从 Nanopore 直接 RNA 测序信号中识别 RNA 修饰位点。
  \item \textbf{TAPE：组织自适应转录组反卷积。}负责单样本基因集富集分析（ssGSEA）结果的分析与解释，并参与论文撰写。该研究通过深度自编码器整合 bulk 与单细胞 RNA-seq 数据，发表于 \textit{Nature Communications}（2022）。
  \item \textbf{FIN：基因融合检测。}参与基于 Transformer 的基因融合检测模型开发。
  \item \textbf{青鳉鱼低氧响应研究。}参与 miRNA--mRNA 整合分析，研究低氧条件下与胚胎发育、性别决定相关的基因调控，成果发表于 \textit{Frontiers in Marine Science}（2022）。
\end{itemize}
}
